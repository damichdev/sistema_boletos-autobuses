-- ESQUEMAS

CREATE SCHEMA administracion;

CREATE SCHEMA viajes;

CREATE SCHEMA ventas;

-- TABLAS EN BRUTO

-- ADMINISTRACION

CREATE TABLE administracion.estado_conexion(
id_conexion INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
nombre_conexion TEXT NOT NULL UNIQUE
);

CREATE TABLE administracion.ciudad(
id_ciudad INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
nombre_ciudad TEXT NOT NULL UNIQUE
);

CREATE TABLE administracion.taller(
id_taller INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
id_ciudad INT NOT NULL,
clave TEXT NOT NULL UNIQUE,
nombre_taller TEXT NOT NULL UNIQUE,
direccion TEXT NOT NULL,
activo BOOLEAN NOT NULL DEFAULT TRUE,
CONSTRAINT fk_ciudad_taller FOREIGN KEY (id_ciudad) REFERENCES administracion.ciudad(id_ciudad) ON DELETE RESTRICT ON UPDATE CASCADE
);

CREATE TABLE administracion.terminal(
id_terminal INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
id_conexion INT NOT NULL,
id_ciudad INT NOT NULL UNIQUE,
nombre_terminal TEXT NOT NULL UNIQUE,
clave TEXT NOT NULL UNIQUE,
direccion TEXT NOT NULL,
telefono VARCHAR(10),
activo BOOLEAN NOT NULL DEFAULT TRUE,
CONSTRAINT check_telefono CHECK (telefono ~'^[0-9]{10}$'),
CONSTRAINT fk_conexion_terminal FOREIGN KEY (id_conexion) REFERENCES administracion.estado_conexion(id_conexion) ON DELETE RESTRICT ON UPDATE CASCADE,
CONSTRAINT fk_ciudad_terminal FOREIGN KEY (id_ciudad) REFERENCES administracion.ciudad(id_ciudad) ON DELETE RESTRICT ON UPDATE CASCADE
);



CREATE TABLE administracion.turno (
    id_turno INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    nombre_turno TEXT NOT NULL UNIQUE
);



CREATE TABLE administracion.taquilla(
id_taquilla INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
id_terminal INT NOT NULL,
identificador TEXT NOT NULL UNIQUE,
rango_inicial INT NOT NULL,
rango_final INT NOT NULL,
serie_folio TEXT NOT NULL UNIQUE,
folios_usados INT NOT NULL DEFAULT 0,
activo BOOLEAN NOT NULL DEFAULT TRUE,
CONSTRAINT check_folios_en_rango CHECK (folios_usados <= rango_final - rango_inicial + 1),
CONSTRAINT check_rangos CHECK (rango_inicial <= rango_final),
CONSTRAINT fk_terminal_taquilla FOREIGN KEY (id_terminal) REFERENCES administracion.terminal(id_terminal) ON DELETE CASCADE
);


CREATE TABLE administracion.puesto(
id_puesto INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
nombre_puesto TEXT NOT NULL UNIQUE
);


CREATE TABLE administracion.personal(
id_personal INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
id_puesto INT NOT NULL,
id_terminal INT,
id_taquilla INT,
id_taller INT,
nombre_personal TEXT NOT NULL,
ap_paterno TEXT NOT NULL,
ap_materno TEXT NOT NULL,
curp TEXT NOT NULL UNIQUE,
telefono VARCHAR(10),
licencia TEXT,
vigencia DATE,
activo BOOLEAN NOT NULL DEFAULT TRUE,
CONSTRAINT check_curp_valido CHECK (curp ~ '^[A-Z]{4}[0-9]{6}[HM][A-Z]{5}[A-Z0-9][0-9]$'),
CONSTRAINT check_telefono CHECK (telefono ~'^[0-9]{10}$'),
CONSTRAINT fk_puesto_personal FOREIGN KEY (id_puesto) REFERENCES administracion.puesto(id_puesto) ON DELETE RESTRICT ON UPDATE CASCADE,
CONSTRAINT fk_terminal_personal FOREIGN KEY (id_terminal) REFERENCES administracion.terminal(id_terminal) ON DELETE RESTRICT ON UPDATE CASCADE,
CONSTRAINT fk_taquilla_personal FOREIGN KEY (id_taquilla) REFERENCES administracion.taquilla(id_taquilla) ON DELETE RESTRICT ON UPDATE CASCADE,
CONSTRAINT fk_taller_personal FOREIGN KEY (id_taller) REFERENCES administracion.taller(id_taller) ON DELETE RESTRICT ON UPDATE CASCADE
);


CREATE TABLE administracion.rol(
id_rol INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
nombre_rol TEXT NOT NULL UNIQUE
);


CREATE UNIQUE INDEX unq_jefe_taller_activo ON administracion.personal (id_taller) WHERE id_puesto = 6 AND activo = TRUE;



CREATE TABLE administracion.personal_app(
id_app INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
id_personal INT NOT NULL UNIQUE,
id_rol INT NOT NULL,
username TEXT NOT NULL UNIQUE,
password_hash TEXT NOT NULL,
activo BOOLEAN NOT NULL DEFAULT TRUE,
CONSTRAINT fk_personal_app FOREIGN KEY (id_personal) REFERENCES administracion.personal(id_personal) ON DELETE CASCADE,
CONSTRAINT fk_rol_app FOREIGN KEY (id_rol) REFERENCES administracion.rol(id_rol) ON DELETE RESTRICT ON UPDATE CASCADE
);



-- VIAJES


CREATE TABLE viajes.nombre_tipo_bus(
id_nombre_tipo_bus INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
nombre_tipo_bus TEXT NOT NULL UNIQUE
);


CREATE TABLE viajes.layout(
id_layout INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
nombre_layout VARCHAR(3) NOT NULL UNIQUE,
CONSTRAINT check_formato_layout CHECK (nombre_layout ~ '^[1-4]\+[1-4]$')
);


CREATE TABLE viajes.tipo_bus(
id_tipo_bus INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
id_nombre_tipo_bus INT NOT NULL,
id_layout INT NOT NULL,
capacidad INT NOT NULL,
CONSTRAINT check_capacidad_positivo CHECK (capacidad > 0),
CONSTRAINT fk_nombre_tipo FOREIGN KEY (id_nombre_tipo_bus) REFERENCES viajes.nombre_tipo_bus(id_nombre_tipo_bus) ON DELETE RESTRICT ON UPDATE CASCADE,
CONSTRAINT fk_layout FOREIGN KEY (id_layout) REFERENCES viajes.layout(id_layout) ON DELETE RESTRICT ON UPDATE CASCADE
);


CREATE TABLE viajes.unidad(
id_unidad INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
id_tipo_bus INT NOT NULL,
placa TEXT NOT NULL UNIQUE,
modelo TEXT NOT NULL,
anio INT NOT NULL,
kilometraje INT NOT NULL,
activo BOOLEAN NOT NULL DEFAULT TRUE,
CONSTRAINT check_anio_futuro CHECK (anio <= EXTRACT(YEAR FROM CURRENT_DATE) +1),
CONSTRAINT check_anio_pasado CHECK (anio > 1900),
CONSTRAINT check_km_positivo CHECK (kilometraje >= 0),
CONSTRAINT fk_tipo_unidad FOREIGN KEY (id_tipo_bus) REFERENCES viajes.tipo_bus(id_tipo_bus) ON DELETE RESTRICT ON UPDATE CASCADE
);


CREATE TABLE viajes.tipo_servicio(
id_servicio INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
nombre_servicio TEXT NOT NULL UNIQUE
);


CREATE TABLE viajes.ruta(
id_ruta INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
id_servicio INT NOT NULL,
nombre_ruta TEXT NOT NULL UNIQUE,
CONSTRAINT fk_servicio_ruta FOREIGN KEY(id_servicio) REFERENCES viajes.tipo_servicio(id_servicio) ON DELETE RESTRICT ON UPDATE CASCADE
);


CREATE TABLE viajes.horario_ruta(
id_ruta INT NOT NULL,
hora_salida TIME NOT NULL,
CONSTRAINT pk_horario_ruta PRIMARY KEY (id_ruta, hora_salida),
CONSTRAINT fk_ruta_horario FOREIGN KEY (id_ruta) REFERENCES viajes.ruta(id_ruta) ON DELETE CASCADE
);


CREATE TABLE viajes.ruta_paradas(
id_parada INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
id_ruta INT NOT NULL,
id_terminal INT NOT NULL,
orden INT NOT NULL,
min_desde_origen INT NOT NULL,
km_desde_origen INT NOT NULL,
CONSTRAINT unique_ruta_orden UNIQUE (id_ruta, orden),
CONSTRAINT unique_ruta_terminal UNIQUE (id_ruta, id_terminal),
CONSTRAINT check_min_positivo CHECK (min_desde_origen >= 0),
CONSTRAINT check_km_positivo CHECK (km_desde_origen >= 0),
CONSTRAINT fk_ruta_parada FOREIGN KEY (id_ruta) REFERENCES viajes.ruta(id_ruta) ON DELETE CASCADE,
CONSTRAINT fk_terminal_parada FOREIGN KEY (id_terminal) REFERENCES administracion.terminal(id_terminal) ON DELETE RESTRICT ON UPDATE CASCADE
);



CREATE TABLE viajes.estado_viaje(
id_estado_viaje INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
nombre_estado_viaje TEXT NOT NULL UNIQUE
);


CREATE TABLE viajes.viaje_programado(
id_viaje INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
id_ruta INT NOT NULL,
id_estado_viaje INT NOT NULL,
id_unidad INT NOT NULL,
id_conductor INT NOT NULL,
id_relevo INT,
fecha_salida TIMESTAMPTZ NOT NULL,
CONSTRAINT check_conductor_relevo CHECK (id_relevo IS NULL OR id_relevo <> id_conductor),
CONSTRAINT fk_ruta_viaje FOREIGN KEY (id_ruta) REFERENCES viajes.ruta(id_ruta) ON DELETE RESTRICT ON UPDATE CASCADE,
CONSTRAINT fk_estado_viaje FOREIGN KEY (id_estado_viaje) REFERENCES viajes.estado_viaje(id_estado_viaje) ON DELETE RESTRICT ON UPDATE CASCADE,
CONSTRAINT fk_unidad_viaje FOREIGN KEY (id_unidad) REFERENCES viajes.unidad(id_unidad) ON DELETE RESTRICT ON UPDATE CASCADE,
CONSTRAINT fk_conductor_viaje FOREIGN KEY (id_conductor) REFERENCES administracion.personal(id_personal) ON DELETE RESTRICT ON UPDATE CASCADE,
CONSTRAINT fk_relevo_viaje FOREIGN KEY (id_relevo) REFERENCES administracion.personal(id_personal) ON DELETE SET NULL
);



CREATE INDEX idx_viaje_fecha_salida ON viajes.viaje_programado (fecha_salida);


CREATE TABLE viajes.precio_tramo(
id_ruta INT NOT NULL,
orden_origen INT NOT NULL,
orden_destino INT NOT NULL,
precio DECIMAL(10,2) NOT NULL,
CONSTRAINT pk_precio_tramo PRIMARY KEY (id_ruta, orden_origen, orden_destino),
CONSTRAINT check_tramo_orden CHECK (orden_origen < orden_destino),
CONSTRAINT check_precio_tramo CHECK (precio >= 0),
CONSTRAINT fk_parada_origen FOREIGN KEY (id_ruta, orden_origen) REFERENCES viajes.ruta_paradas(id_ruta, orden) ON DELETE CASCADE ON UPDATE CASCADE,
CONSTRAINT fk_parada_destino FOREIGN KEY (id_ruta, orden_destino) REFERENCES viajes.ruta_paradas(id_ruta, orden) ON DELETE CASCADE ON UPDATE CASCADE
);


-- CLIENTES


CREATE TABLE ventas.pasajero(
id_pasajero INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
nombre_pasajero TEXT NOT NULL,
ap_paterno TEXT NOT NULL,
ap_materno TEXT NOT NULL,
curp TEXT UNIQUE,
CONSTRAINT check_curp_valido CHECK (curp ~ '^[A-Z]{4}[0-9]{6}[HMX][A-Z]{5}[A-Z0-9][0-9]$')
);


CREATE TABLE ventas.estado_boleto(
id_estado_boleto INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
nombre_estado_boleto TEXT NOT NULL UNIQUE
);


CREATE TABLE ventas.tarifa(
id_tarifa INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
nombre_tarifa TEXT NOT NULL UNIQUE,
descuento NUMERIC(4,3) NOT NULL DEFAULT 0,
CONSTRAINT check_descuento CHECK (descuento >= 0 AND descuento <= 1)
);


CREATE TABLE ventas.venta(
id_venta INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
id_taquilla INT NOT NULL,
folio INT NOT NULL,
id_titular INT NOT NULL,
id_vendedor INT NOT NULL,
id_turno INT NOT NULL,
fecha TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
total_pagado DECIMAL(10,2) NOT NULL,
reemplaza_a INT,
recibido DECIMAL(10,2) NOT NULL,
cancelada_por INT,
cancelada_en TIMESTAMPTZ,
CONSTRAINT unique_folio UNIQUE (id_taquilla, folio),
CONSTRAINT check_pagado_positivo CHECK (total_pagado >= 0),
CONSTRAINT check_recibido CHECK (recibido >= total_pagado),
CONSTRAINT unique_reemplaza_a UNIQUE (reemplaza_a),
CONSTRAINT check_cancelacion CHECK ((cancelada_por IS NULL) = (cancelada_en IS NULL)),
CONSTRAINT fk_turno_venta FOREIGN KEY (id_turno) REFERENCES administracion.turno(id_turno) ON DELETE RESTRICT ON UPDATE CASCADE,
CONSTRAINT fk_taquilla_venta FOREIGN KEY (id_taquilla) REFERENCES administracion.taquilla(id_taquilla) ON DELETE RESTRICT ON UPDATE CASCADE,
CONSTRAINT fk_pasajero_venta FOREIGN KEY (id_titular) REFERENCES ventas.pasajero(id_pasajero) ON DELETE RESTRICT ON UPDATE CASCADE,
CONSTRAINT fk_personal_venta FOREIGN KEY (id_vendedor) REFERENCES administracion.personal_app(id_app) ON DELETE RESTRICT ON UPDATE CASCADE,
CONSTRAINT fk_reemplaza_venta FOREIGN KEY (reemplaza_a) REFERENCES ventas.venta(id_venta) ON DELETE RESTRICT,
CONSTRAINT fk_cancelada_por FOREIGN KEY (cancelada_por) REFERENCES administracion.personal_app(id_app) ON DELETE RESTRICT
);


CREATE TABLE ventas.boleto(
id_boleto INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
id_venta INT NOT NULL,
id_viaje INT NOT NULL,
id_terminal_origen INT NOT NULL,
id_terminal_destino INT NOT NULL,
num_asiento INT NOT NULL,
tramo_ocupacion int4range NOT NULL,
id_tarifa INT NOT NULL,
id_estado_boleto INT NOT NULL,
precio_unitario DECIMAL(10,2) NOT NULL,
CONSTRAINT check_asiento_positivo CHECK (num_asiento > 0),
CONSTRAINT check_precio_positivo CHECK (precio_unitario >= 0),
CONSTRAINT fk_venta_boleto FOREIGN KEY (id_venta) REFERENCES ventas.venta(id_venta) ON DELETE RESTRICT ON UPDATE CASCADE,
CONSTRAINT fk_viaje_boleto FOREIGN KEY (id_viaje) REFERENCES viajes.viaje_programado(id_viaje) ON DELETE RESTRICT ON UPDATE CASCADE,
CONSTRAINT fk_terminal_origen_boleto FOREIGN KEY (id_terminal_origen) REFERENCES administracion.terminal(id_terminal) ON DELETE RESTRICT ON UPDATE CASCADE,
CONSTRAINT fk_terminal_destino_boleto FOREIGN KEY (id_terminal_destino) REFERENCES administracion.terminal(id_terminal) ON DELETE RESTRICT ON UPDATE CASCADE,
CONSTRAINT fk_tarifa_boleto FOREIGN KEY (id_tarifa) REFERENCES ventas.tarifa(id_tarifa) ON DELETE RESTRICT ON UPDATE CASCADE,
CONSTRAINT fk_estado_boleto FOREIGN KEY (id_estado_boleto) REFERENCES ventas.estado_boleto(id_estado_boleto) ON DELETE RESTRICT ON UPDATE CASCADE,
CONSTRAINT evitar_sobreventa EXCLUDE USING gist (id_viaje WITH =, num_asiento WITH =, tramo_ocupacion WITH &&) WHERE (id_estado_boleto <> 4)
);

-- VISTAS ADMINISTRACION

CREATE OR REPLACE VIEW administracion.v_terminales AS 
SELECT id_terminal, nombre_terminal
FROM administracion.terminal
WHERE activo = TRUE
ORDER BY nombre_terminal;



CREATE OR REPLACE VIEW administracion.v_directorio_usuarios AS
SELECT
pa.username AS usuario,
p.nombre_personal || ' ' || p.ap_paterno || ' ' || p.ap_materno AS nombre_personal,
r.nombre_rol AS rol,
pu.nombre_puesto AS puesto,
COALESCE(tq.identificador, 'SIN CAJA ASIGNADA') AS caja_asignada,
COALESCE(term.nombre_terminal, 'SIN TERMINAL OPERATIVA') AS ubicacion
FROM administracion.personal_app pa
INNER JOIN administracion.personal p ON pa.id_personal = p.id_personal
INNER JOIN administracion.rol r ON r.id_rol = pa.id_rol
INNER JOIN administracion.puesto pu ON pu.id_puesto = p.id_puesto
LEFT JOIN administracion.taquilla tq ON p.id_taquilla = tq.id_taquilla
LEFT JOIN administracion.terminal term ON p.id_terminal = term.id_terminal;



CREATE OR REPLACE VIEW administracion.v_terminales_admin AS
SELECT t.id_terminal, t.nombre_terminal, t.activo
FROM administracion.terminal t
ORDER BY t.nombre_terminal;



CREATE OR REPLACE VIEW administracion.v_roles AS
SELECT id_rol, nombre_rol 
FROM administracion.rol 
ORDER BY id_rol;



CREATE OR REPLACE VIEW administracion.v_cuentas AS
SELECT 
pa.id_app, 
pa.id_personal, 
pa.username,
p.nombre_personal || ' ' || p.ap_paterno || COALESCE(' ' || p.ap_materno, '') AS nombre_completo,
r.id_rol, 
r.nombre_rol, 
t.id_terminal, 
t.nombre_terminal, 
q.id_taquilla, 
q.identificador, 
pa.activo,
tl.id_taller,
tl.nombre_taller
FROM administracion.personal_app pa
INNER JOIN administracion.personal p ON p.id_personal = pa.id_personal
INNER JOIN administracion.rol r ON r.id_rol = pa.id_rol
LEFT JOIN administracion.terminal t ON t.id_terminal = p.id_terminal
LEFT JOIN administracion.taquilla q ON q.id_taquilla = p.id_taquilla
LEFT JOIN administracion.taller tl ON tl.id_taller = p.id_taller
ORDER BY pa.username;



CREATE OR REPLACE VIEW administracion.v_puestos AS
SELECT id_puesto, nombre_puesto 
FROM administracion.puesto 
ORDER BY id_puesto;



CREATE OR REPLACE VIEW administracion.v_personal AS
SELECT 
p.id_personal, 
pu.id_puesto, 
pu.nombre_puesto, 
p.nombre_personal, 
p.ap_paterno, 
p.ap_materno,
p.curp, 
p.telefono, 
p.licencia, 
p.vigencia,
EXISTS (SELECT 1 FROM administracion.personal_app pa WHERE pa.id_personal = p.id_personal) AS tiene_cuenta,
p.id_taller,
p.id_terminal,
p.id_taquilla
FROM administracion.personal p
INNER JOIN administracion.puesto pu ON pu.id_puesto = p.id_puesto
ORDER BY p.ap_paterno, p.ap_materno, p.nombre_personal;



CREATE OR REPLACE VIEW administracion.v_talleres_activos AS
SELECT id_taller, nombre_taller
FROM administracion.taller
WHERE activo
ORDER BY nombre_taller;



CREATE OR REPLACE VIEW administracion.v_talleres AS
SELECT
t.id_taller,
t.id_ciudad,
c.nombre_ciudad,
t.clave,
t.nombre_taller,
t.direccion,
t.activo,
(SELECT COUNT(*) FROM administracion.personal p WHERE p.id_taller = t.id_taller AND p.id_puesto = 4 AND p.activo)::INT AS mecanicos,
(SELECT p.nombre_personal || ' ' || p.ap_paterno FROM administracion.personal p
 WHERE p.id_taller = t.id_taller AND p.id_puesto = 6 AND p.activo) AS encargado
FROM administracion.taller t
INNER JOIN administracion.ciudad c ON c.id_ciudad = t.id_ciudad
ORDER BY c.nombre_ciudad, t.nombre_taller;



CREATE OR REPLACE VIEW administracion.v_choferes AS
SELECT
p.id_personal,
p.nombre_personal || ' ' || p.ap_paterno || COALESCE(' ' || p.ap_materno, '') AS chofer
FROM administracion.personal p
INNER JOIN administracion.puesto pu ON pu.id_puesto = p.id_puesto
WHERE nombre_puesto = 'CONDUCTOR'
ORDER BY p.ap_paterno, p.ap_materno, p.nombre_personal;



CREATE OR REPLACE VIEW administracion.v_taquillas AS
SELECT 
q.id_taquilla, 
t.id_terminal, 
t.nombre_terminal, 
q.identificador, 
q.serie_folio, 
q.rango_inicial, 
q.rango_final, 
q.folios_usados, 
q.activo
FROM administracion.taquilla q
INNER JOIN administracion.terminal t ON t.id_terminal = q.id_terminal
ORDER BY t.nombre_terminal, q.identificador;



CREATE OR REPLACE VIEW administracion.v_ciudades AS
SELECT id_ciudad, nombre_ciudad
FROM administracion.ciudad
ORDER BY nombre_ciudad;



CREATE OR REPLACE VIEW administracion.v_terminales_detalle AS
SELECT t.id_terminal, t.nombre_terminal, t.activo, c.id_ciudad, c.nombre_ciudad, t.clave, t.direccion, t.telefono
FROM administracion.terminal t
INNER JOIN administracion.ciudad c ON c.id_ciudad = t.id_ciudad
ORDER BY c.nombre_ciudad;



-- VISTAS VIAJES


CREATE OR REPLACE VIEW viajes.cartelera_viajes AS
SELECT
v.id_viaje,
r.nombre_ruta AS ruta,
ts.nombre_servicio AS servicio,
v.fecha_salida,
ev.nombre_estado_viaje AS estado_viaje,
u.placa AS unidad,
u.modelo,
per.nombre_personal || ' ' || per.ap_paterno || ' ' || per.ap_materno AS conductor,

CREATE FUNCTION administracion.fn_validar_personal() RETURNS TRIGGER
LANGUAGE plpgsql AS $$
DECLARE
  v_puesto TEXT;
  v_tiene_jefe_cuenta BOOLEAN;
BEGIN
  SELECT nombre_puesto INTO v_puesto FROM administracion.puesto WHERE id_puesto = NEW.id_puesto;
  IF v_puesto = 'CONDUCTOR' THEN
    IF NEW.licencia IS NULL OR btrim(NEW.licencia) = '' OR NEW.vigencia IS NULL THEN
      RAISE EXCEPTION 'El conductor necesita licencia y vigencia.' USING ERRCODE = 'Z0021';
    END IF;
  ELSIF NEW.licencia IS NOT NULL OR NEW.vigencia IS NOT NULL THEN
    RAISE EXCEPTION 'Solo el conductor lleva licencia y vigencia.' USING ERRCODE = 'Z0021';
  END IF;
  IF NEW.id_terminal IS NOT NULL AND v_puesto NOT IN ('TAQUILLERO','DESPACHADOR') THEN
    RAISE EXCEPTION 'Solo taquilleros y despachadores se asignan a una terminal.' USING ERRCODE = 'Z0022';
  END IF;
  IF NEW.id_taquilla IS NOT NULL AND v_puesto <> 'TAQUILLERO' THEN
    RAISE EXCEPTION 'Solo los taquilleros se asignan a una taquilla.' USING ERRCODE = 'Z0023';
  END IF;
  IF NEW.id_taller IS NOT NULL AND v_puesto NOT IN ('MECANICO','JEFE_TALLER') THEN
    RAISE EXCEPTION 'Solo mecánicos y jefes de taller se asignan a un taller.' USING ERRCODE = 'Z0024';
  END IF;
IF TG_OP = 'UPDATE' AND v_puesto = 'JEFE_TALLER' AND NEW.id_taller IS NULL THEN
    SELECT EXISTS (SELECT 1 FROM administracion.personal_app WHERE id_personal = NEW.id_personal) INTO v_tiene_jefe_cuenta;
    IF v_tiene_jefe_cuenta THEN
      RAISE EXCEPTION 'El jefe de taller tiene cuenta: debe conservar su taller.' USING ERRCODE = 'Z0025';
    END IF;
  END IF;
IF NEW.id_taquilla IS NOT NULL THEN
    IF (SELECT id_terminal FROM administracion.taquilla WHERE id_taquilla = NEW.id_taquilla) <> NEW.id_terminal THEN
      RAISE EXCEPTION 'La taquilla seleccionada no pertenece a la terminal asignada.' USING ERRCODE = 'Z0061';
    END IF;
  END IF;
  RETURN NEW;
END $$;


CREATE TRIGGER tigresito_validar_personal
BEFORE INSERT OR UPDATE ON administracion.personal
FOR EACH ROW EXECUTE FUNCTION administracion.fn_validar_personal();




CREATE FUNCTION administracion.fn_validar_cuenta() RETURNS TRIGGER
LANGUAGE plpgsql AS $$
DECLARE
  v_puesto TEXT;
  v_rol TEXT;
  v_taller INT;
BEGIN
  SELECT pu.nombre_puesto, p.id_taller INTO v_puesto, v_taller
  FROM administracion.personal p JOIN administracion.puesto pu ON pu.id_puesto = p.id_puesto
  WHERE p.id_personal = NEW.id_personal;
  SELECT nombre_rol INTO v_rol FROM administracion.rol WHERE id_rol = NEW.id_rol;
  IF v_puesto IN ('CONDUCTOR','MECANICO') THEN
    RAISE EXCEPTION 'Choferes y mecánicos no tienen cuenta.' USING ERRCODE = 'Z0026';
  END IF;
  IF v_rol IS DISTINCT FROM v_puesto THEN
    RAISE EXCEPTION 'El rol (%) no corresponde al puesto (%).', v_rol, v_puesto USING ERRCODE = 'Z0027';
  END IF;
  IF v_puesto = 'JEFE_TALLER' AND v_taller IS NULL THEN
    RAISE EXCEPTION 'Asigne un taller al jefe antes de crear su cuenta.' USING ERRCODE = 'Z0025';
  END IF;
  RETURN NEW;
END $$;


CREATE TRIGGER tigresito_validar_cuenta
BEFORE INSERT OR UPDATE OF id_personal, id_rol ON administracion.personal_app
FOR EACH ROW EXECUTE FUNCTION administracion.fn_validar_cuenta();




CREATE OR REPLACE FUNCTION viajes.fn_asignar_conductor()
RETURNS TRIGGER AS $$
DECLARE
    v_chofer INT;
    v_puesto TEXT;
    v_licencia TEXT;
    v_vigencia DATE;
    v_duracion INT;
    v_fin TIMESTAMPTZ;
BEGIN
    IF NOT EXISTS (SELECT 1 FROM viajes.unidad WHERE id_unidad = NEW.id_unidad AND activo = TRUE) THEN
        RAISE EXCEPTION 'La unidad asignada se encuentra inactiva o dada de baja.' USING ERRCODE = 'Z0063';
    END IF;
    FOREACH v_chofer IN ARRAY ARRAY_REMOVE(ARRAY[NEW.id_conductor, NEW.id_relevo], NULL) LOOP
        SELECT pu.nombre_puesto, p.licencia, p.vigencia INTO v_puesto, v_licencia, v_vigencia
        FROM administracion.personal p
        INNER JOIN administracion.puesto pu ON p.id_puesto = pu.id_puesto
        WHERE p.id_personal = v_chofer;
        IF v_puesto <> 'CONDUCTOR' THEN
            RAISE EXCEPTION 'El empleado % no es CONDUCTOR.', v_chofer USING ERRCODE = 'Z0007';
        END IF;
        IF v_licencia IS NULL OR v_vigencia IS NULL OR v_vigencia < (NEW.fecha_salida AT TIME ZONE 'America/Mexico_City')::DATE THEN
            RAISE EXCEPTION 'La licencia del conductor % no está vigente el día del viaje.', v_chofer USING ERRCODE = 'Z0006';
        END IF;
    END LOOP;
    IF NEW.id_estado_viaje IN (1, 2, 3) THEN
        SELECT MAX(min_desde_origen) INTO v_duracion FROM viajes.ruta_paradas WHERE id_ruta = NEW.id_ruta;
        v_fin := NEW.fecha_salida + make_interval(mins => COALESCE(v_duracion, 0));
        IF EXISTS (
            SELECT 1
            FROM viajes.viaje_programado o
            WHERE o.id_viaje <> COALESCE(NEW.id_viaje, -1)
            AND o.id_estado_viaje IN (1, 2, 3)
            AND (o.id_unidad = NEW.id_unidad
            OR o.id_conductor IN (NEW.id_conductor, NEW.id_relevo)
            OR o.id_relevo IN (NEW.id_conductor, NEW.id_relevo))
            AND tstzrange(o.fecha_salida, o.fecha_salida + make_interval(mins => COALESCE(
            (SELECT MAX(rp.min_desde_origen) FROM viajes.ruta_paradas rp WHERE rp.id_ruta = o.id_ruta), 0)), '[)')
            && tstzrange(NEW.fecha_salida, v_fin, '[)')
        ) THEN
            RAISE EXCEPTION 'La unidad o el conductor ya tienen otro viaje en ese horario.' USING ERRCODE = 'Z0012';
        END IF;
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;


CREATE TRIGGER tigresito_validad_conductor
BEFORE INSERT OR UPDATE ON viajes.viaje_programado
FOR EACH ROW EXECUTE FUNCTION viajes.fn_asignar_conductor();



CREATE OR REPLACE FUNCTION viajes.fn_validar_parada()
RETURNS TRIGGER AS $$
DECLARE
v_prev_min INT;
v_prev_km INT;
BEGIN
IF NEW.orden = 1 THEN
IF NEW.min_desde_origen <> 0 OR NEW.km_desde_origen <> 0 THEN
RAISE EXCEPTION 'La primera parada debe tener 0 minutos y 0 km desde el origen.' USING ERRCODE = 'Z0067';
END IF;
ELSE
SELECT min_desde_origen, km_desde_origen INTO v_prev_min, v_prev_km
FROM viajes.ruta_paradas
WHERE id_ruta = NEW.id_ruta AND orden = NEW.orden -1;
IF NOT FOUND THEN
RAISE EXCEPTION 'Las paradas deben guardarse en orden consecutivo (falta la parada %).', NEW.orden - 1 USING ERRCODE = 'Z0067';
END IF;
IF NEW.min_desde_origen <= v_prev_min OR NEW.km_desde_origen < v_prev_km THEN
RAISE EXCEPTION 'Los minutos y los kilómetros deben aumentar de una parada a la siguiente.' USING ERRCODE = 'Z0067';
END IF;
END IF;
RETURN NEW;
END;
$$ LANGUAGE plpgsql;


CREATE TRIGGER tigresito_validar_parada
BEFORE INSERT ON viajes.ruta_paradas
FOR EACH ROW EXECUTE FUNCTION viajes.fn_validar_parada();



CREATE OR REPLACE FUNCTION ventas.fn_preparar_boleto()
RETURNS TRIGGER AS $$
DECLARE
v_id_ruta INT;
v_capacidad_bus INT;
v_orden_origen INT;
v_orden_destino INT;
BEGIN
SELECT vp.id_ruta, tb.capacidad INTO v_id_ruta, v_capacidad_bus
FROM viajes.viaje_programado vp
INNER JOIN viajes.unidad u ON vp.id_unidad = u.id_unidad
INNER JOIN viajes.tipo_bus tb ON tb.id_tipo_bus = u.id_tipo_bus
WHERE vp.id_viaje = NEW.id_viaje;
IF NEW.num_asiento > v_capacidad_bus THEN
RAISE EXCEPTION 'El asiento % no existe, esta unidad solo tiene capacidad para % pasajeros a bordo.', NEW.num_asiento, v_capacidad_bus USING ERRCODE = 'Z0013';
END IF;       
SELECT orden INTO v_orden_origen
FROM viajes.ruta_paradas
WHERE id_ruta = v_id_ruta AND id_terminal = NEW.id_terminal_origen;    
IF NOT FOUND THEN
RAISE EXCEPTION 'La terminal de origen seleccionada no pertenece a la ruta de este viaje.' USING ERRCODE = 'Z0014';
END IF;    
SELECT orden INTO v_orden_destino
FROM viajes.ruta_paradas
WHERE id_ruta = v_id_ruta AND id_terminal = NEW.id_terminal_destino;    
IF NOT FOUND THEN
RAISE EXCEPTION 'La terminal de destino seleccionada no pertenece a la ruta de este viaje.' USING ERRCODE = 'Z0015';
END IF;    
IF v_orden_origen >= v_orden_destino THEN
RAISE EXCEPTION 'El destino del pasajero debe ser despues del punto de origen.' USING ERRCODE = 'Z0016';
END IF;    
NEW.tramo_ocupacion := int4range(v_orden_origen, v_orden_destino, '[)');
RETURN NEW;
END;
$$ LANGUAGE plpgsql;


CREATE TRIGGER tigresito_preparar_boleto
BEFORE INSERT OR UPDATE ON ventas.boleto
FOR EACH ROW EXECUTE FUNCTION ventas.fn_preparar_boleto();




CREATE OR REPLACE FUNCTION ventas.fn_validar_folio()
RETURNS TRIGGER AS $$
DECLARE
    v_inicial INT;
    v_final INT;
BEGIN
    SELECT rango_inicial, rango_final INTO v_inicial, v_final
    FROM administracion.taquilla
    WHERE id_taquilla = NEW.id_taquilla;
    IF NEW.folio < v_inicial OR NEW.folio > v_final THEN
        RAISE EXCEPTION 'El folio % está fuera del rango de la taquilla (% a %).', NEW.folio, v_inicial, v_final USING ERRCODE = 'Z0002';
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;


CREATE TRIGGER tigresito_validar_folio
BEFORE INSERT OR UPDATE OF folio, id_taquilla ON ventas.venta
FOR EACH ROW EXECUTE FUNCTION ventas.fn_validar_folio();

CREATE OR REPLACE FUNCTION viajes.fn_buscar_salidas(p_id_origen INT, p_id_destino INT, p_fecha DATE)
RETURNS TABLE (id_viaje INT, fecha_salida TIMESTAMPTZ, tipo_bus TEXT, servicio TEXT, placa TEXT, libres INT, precio DECIMAL)
LANGUAGE sql
AS $$
SELECT
vp.id_viaje,
vp.fecha_salida + make_interval(mins => o.min_desde_origen),
ntb.nombre_tipo_bus,
ts.nombre_servicio,
u.placa,
tb.capacidad - (SELECT count(*) FROM ventas.boleto b
                WHERE b.id_viaje = vp.id_viaje
                AND b.id_estado_boleto <> 4
                AND b.tramo_ocupacion && int4range(o.orden, d.orden, '[)'))::INT,
pt.precio
FROM viajes.viaje_programado vp
INNER JOIN viajes.estado_viaje ev ON ev.id_estado_viaje = vp.id_estado_viaje
INNER JOIN viajes.ruta r ON r.id_ruta = vp.id_ruta
INNER JOIN viajes.tipo_servicio ts ON ts.id_servicio = r.id_servicio
INNER JOIN viajes.unidad u ON u.id_unidad = vp.id_unidad
INNER JOIN viajes.tipo_bus tb ON tb.id_tipo_bus = u.id_tipo_bus
INNER JOIN viajes.nombre_tipo_bus ntb ON ntb.id_nombre_tipo_bus = tb.id_nombre_tipo_bus
INNER JOIN viajes.ruta_paradas o ON o.id_ruta = vp.id_ruta AND o.id_terminal = p_id_origen
INNER JOIN viajes.ruta_paradas d ON d.id_ruta = vp.id_ruta AND d.id_terminal = p_id_destino AND d.orden > o.orden
INNER JOIN viajes.precio_tramo pt ON pt.id_ruta = vp.id_ruta AND pt.orden_origen = o.orden AND pt.orden_destino = d.orden
WHERE (ev.nombre_estado_viaje IN ('PROGRAMADO', 'ABORDANDO')
       OR (ev.nombre_estado_viaje = 'EN RUTA'
           AND o.orden > (SELECT min(rp.orden) FROM viajes.ruta_paradas rp WHERE rp.id_ruta = vp.id_ruta)))
AND vp.fecha_salida + make_interval(mins => o.min_desde_origen) > now()
AND ((vp.fecha_salida + make_interval(mins => o.min_desde_origen)) AT TIME ZONE 'America/Mexico_City')::date = p_fecha
ORDER BY 2;
$$;



CREATE OR REPLACE FUNCTION viajes.fn_obtener_escalas(p_id_ruta INT)
RETURNS TABLE (
id_terminal INT, 
nombre_terminal TEXT, 
min_desde_origen INT
) 
LANGUAGE sql
AS $$
SELECT t.id_terminal, t.nombre_terminal, rp.min_desde_origen
FROM viajes.ruta_paradas rp
INNER JOIN administracion.terminal t ON t.id_terminal = rp.id_terminal
WHERE rp.id_ruta = p_id_ruta 
AND rp.orden < (SELECT max(orden) FROM viajes.ruta_paradas WHERE id_ruta = p_id_ruta)
ORDER BY rp.orden;
$$;



CREATE OR REPLACE FUNCTION viajes.fn_buscar_salidas_por_dia(p_inicio TIMESTAMPTZ, p_fin TIMESTAMPTZ)
RETURNS TABLE (

placa TEXT, modelo TEXT, capacidad INT, 
nombre_conductor TEXT, vigencia DATE
) 
LANGUAGE sql
AS $$
SELECT * FROM viajes.v_salidas 
WHERE fecha_salida >= p_inicio AND fecha_salida < p_fin
ORDER BY fecha_salida;
$$;



CREATE OR REPLACE FUNCTION viajes.fn_obtener_recorrido_viaje(p_id_viaje INT, p_id_ruta INT)
RETURNS TABLE (
nombre_terminal TEXT, 
min_desde_origen INT, 
suben BIGINT, 
bajan BIGINT, 
ocupacion_en_tramo BIGINT
) 
LANGUAGE sql
AS $$
SELECT 
t.nombre_terminal, 
rp.min_desde_origen,
(SELECT count(*) FROM ventas.boleto b WHERE b.id_viaje = p_id_viaje AND b.id_estado_boleto <> 4 AND b.id_terminal_origen = rp.id_terminal) AS suben,
(SELECT count(*) FROM ventas.boleto b WHERE b.id_viaje = p_id_viaje AND b.id_estado_boleto <> 4 AND b.id_terminal_destino = rp.id_terminal) AS bajan,
(SELECT count(*) FROM ventas.boleto b WHERE b.id_viaje = p_id_viaje AND b.id_estado_boleto <> 4 AND b.tramo_ocupacion @> rp.orden) AS ocupacion_en_tramo
FROM viajes.ruta_paradas rp
INNER JOIN administracion.terminal t ON t.id_terminal = rp.id_terminal
WHERE rp.id_ruta = p_id_ruta
ORDER BY rp.orden;
$$;

CREATE OR REPLACE FUNCTION viajes.fn_obtener_destinos_validos(p_id_origen INT)
RETURNS TABLE (id_terminal INT, nombre_terminal TEXT)

CREATE OR REPLACE FUNCTION viajes.fn_destinos_por_salida(p_id_origen INT, p_fecha DATE)
RETURNS TABLE (
id_terminal INT,
nombre_terminal TEXT
)
LANGUAGE sql
AS $$
SELECT DISTINCT t.id_terminal, t.nombre_terminal
FROM viajes.viaje_programado vp
INNER JOIN viajes.estado_viaje ev ON ev.id_estado_viaje = vp.id_estado_viaje
INNER JOIN viajes.ruta_paradas o ON o.id_ruta = vp.id_ruta AND o.id_terminal = p_id_origen
INNER JOIN viajes.ruta_paradas d ON d.id_ruta = vp.id_ruta AND d.orden > o.orden
INNER JOIN viajes.precio_tramo pt ON pt.id_ruta = vp.id_ruta AND  pt.orden_origen = o.orden AND pt.orden_destino = d.orden
INNER JOIN administracion.terminal t ON t.id_terminal = d.id_terminal
WHERE (ev.nombre_estado_viaje IN ('PROGRAMADO', 'ABORDANDO')
OR (ev.nombre_estado_viaje = 'EN RUTA' AND o.orden > (SELECT min(rp.orden) FROM viajes.ruta_paradas rp WHERE rp.id_ruta = vp.id_ruta)))
AND vp.fecha_salida + make_interval(mins => o.min_desde_origen) > now()
AND ((vp.fecha_salida + make_interval(mins => o.min_desde_origen))
AT TIME ZONE 'America/Mexico_City')::date = p_fecha
ORDER BY t.nombre_terminal;
$$;

CREATE OR REPLACE FUNCTION ventas.fn_obtener_asientos_ocupados(p_id_viaje INT, p_id_origen INT, p_id_destino INT)
RETURNS TABLE (

)
LANGUAGE sql
AS $$
SELECT b.num_asiento
FROM ventas.boleto b
WHERE b.id_viaje = p_id_viaje
AND b.id_estado_boleto <> 4
AND b.tramo_ocupacion && (
SELECT int4range(o.orden, d.orden, '[)')
FROM viajes.viaje_programado vp
INNER JOIN viajes.ruta_paradas o ON o.id_ruta = vp.id_ruta AND o.id_terminal = p_id_origen
INNER JOIN viajes.ruta_paradas d ON d.id_ruta = vp.id_ruta AND d.id_terminal = p_id_destino
WHERE vp.id_viaje = p_id_viaje
)
ORDER BY b.num_asiento;
$$;

CREATE PROCEDURE administracion.sp_cambiar_estado_terminal(p_id_terminal INT, p_activo BOOLEAN)
LANGUAGE plpgsql SECURITY DEFINER SET search_path = pg_catalog, pg_temp AS $$
BEGIN
  IF NOT p_activo AND EXISTS (
    SELECT 1 FROM viajes.viaje_programado v
    JOIN viajes.ruta_paradas rp ON rp.id_ruta = v.id_ruta
    WHERE rp.id_terminal = p_id_terminal AND v.id_estado_viaje IN (1, 2, 3) AND v.fecha_salida >= now() - interval '1 day'
  ) THEN
    RAISE EXCEPTION 'La terminal tiene salidas programadas o en ruta: reasígnelas antes de desactivarla.' USING ERRCODE = 'Z0030';
  END IF;
  UPDATE administracion.terminal SET activo = p_activo WHERE id_terminal = p_id_terminal;
END $$;



CREATE PROCEDURE administracion.sp_crear_ciudad(INOUT p_id_ciudad INT, p_nombre TEXT)
LANGUAGE plpgsql SECURITY DEFINER SET search_path = pg_catalog, pg_temp AS $$
BEGIN
  p_nombre := upper(btrim(p_nombre));
  IF p_nombre = '' THEN RAISE EXCEPTION 'Escriba el nombre de la ciudad.' USING ERRCODE = 'Z0031'; END IF;
  INSERT INTO administracion.ciudad (nombre_ciudad) VALUES (p_nombre)
  ON CONFLICT (nombre_ciudad) DO NOTHING;
  SELECT id_ciudad INTO p_id_ciudad FROM administracion.ciudad WHERE nombre_ciudad = p_nombre;
END $$;



CREATE PROCEDURE administracion.sp_guardar_terminal(INOUT p_id_terminal INT, p_id_ciudad INT, p_clave TEXT,
    p_nombre TEXT, p_direccion TEXT, p_telefono TEXT)
LANGUAGE plpgsql SECURITY DEFINER SET search_path = pg_catalog, pg_temp AS $$
BEGIN
  IF COALESCE(p_id_terminal, 0) = 0 THEN
    IF EXISTS (SELECT 1 FROM administracion.terminal WHERE id_ciudad = p_id_ciudad) THEN
      RAISE EXCEPTION 'Esa ciudad ya tiene su terminal.' USING ERRCODE = 'Z0032';
    END IF;
    INSERT INTO administracion.terminal (id_conexion, id_ciudad, clave, nombre_terminal, direccion, telefono)
    VALUES (4, p_id_ciudad, upper(btrim(p_clave)), btrim(p_nombre), btrim(p_direccion), NULLIF(btrim(p_telefono), ''))
    RETURNING id_terminal INTO p_id_terminal;
  ELSE
    UPDATE administracion.terminal
    SET clave = upper(btrim(p_clave)), nombre_terminal = btrim(p_nombre), direccion = btrim(p_direccion),
        telefono = NULLIF(btrim(p_telefono), '')
    WHERE id_terminal = p_id_terminal;
    IF NOT FOUND THEN RAISE EXCEPTION 'La terminal no existe.' USING ERRCODE = 'Z0033'; END IF;
  END IF;
END $$;



CREATE PROCEDURE administracion.sp_guardar_taquilla(p_id_taquilla INT, p_id_terminal INT, p_identificador TEXT,
    p_serie_folio TEXT, p_rango_inicial INT, p_rango_final INT, p_activo BOOLEAN)
LANGUAGE plpgsql SECURITY DEFINER SET search_path = pg_catalog, pg_temp AS $$
DECLARE
  v_usados INT;
  v_inicial INT;
BEGIN
  IF p_id_taquilla = 0 THEN
    INSERT INTO administracion.taquilla (id_terminal, identificador, serie_folio, rango_inicial, rango_final, activo)
    VALUES (p_id_terminal, p_identificador, p_serie_folio, p_rango_inicial, p_rango_final, p_activo);
  ELSE
    SELECT folios_usados, rango_inicial INTO v_usados, v_inicial FROM administracion.taquilla WHERE id_taquilla = p_id_taquilla FOR UPDATE;
    IF NOT FOUND THEN RAISE EXCEPTION 'La taquilla no existe.' USING ERRCODE = 'Z0034'; END IF;
    IF v_usados > 0 AND p_rango_inicial <> v_inicial THEN
      RAISE EXCEPTION 'La taquilla ya vendió folios: no se puede cambiar el folio inicial.' USING ERRCODE = 'Z0035';
    END IF;
    IF p_rango_final < p_rango_inicial + v_usados - 1 THEN
      RAISE EXCEPTION 'El folio final no puede quedar por debajo de los folios ya usados.' USING ERRCODE = 'Z0035';
    END IF;
    UPDATE administracion.taquilla
    SET id_terminal = p_id_terminal, identificador = p_identificador, serie_folio = p_serie_folio,
        rango_inicial = p_rango_inicial, rango_final = p_rango_final, activo = p_activo
    WHERE id_taquilla = p_id_taquilla;
  END IF;
END $$;

CREATE PROCEDURE administracion.sp_asignar_encargado(p_id_taller INT, p_id_personal INT)
LANGUAGE plpgsql SECURITY DEFINER SET search_path = pg_catalog, pg_temp AS $$
BEGIN
  IF p_id_personal IS NOT NULL AND NOT EXISTS (
      SELECT 1 FROM administracion.personal WHERE id_personal = p_id_personal AND id_puesto = 6 AND activo) THEN
    RAISE EXCEPTION 'El encargado debe ser un jefe de taller activo.' USING ERRCODE = 'Z0051';
  END IF;
  IF p_id_personal IS NOT NULL AND EXISTS (
      SELECT 1 FROM administracion.personal WHERE id_personal = p_id_personal AND id_taller IS NOT NULL AND id_taller <> p_id_taller) THEN
    RAISE EXCEPTION 'Ese jefe ya está asignado a otro taller.' USING ERRCODE = 'Z0051';
  END IF;
  UPDATE administracion.personal SET id_taller = NULL
  WHERE id_taller = p_id_taller AND id_puesto = 6 AND id_personal IS DISTINCT FROM p_id_personal
    AND NOT EXISTS (SELECT 1 FROM administracion.personal_app pa WHERE pa.id_personal = personal.id_personal);
  IF EXISTS (SELECT 1 FROM administracion.personal j JOIN administracion.personal_app pa ON pa.id_personal = j.id_personal
             WHERE j.id_taller = p_id_taller AND j.id_puesto = 6 AND j.id_personal IS DISTINCT FROM p_id_personal) THEN
    RAISE EXCEPTION 'El encargado actual tiene cuenta: désele de baja o cámbiele de taller antes.' USING ERRCODE = 'Z0051';
  END IF;
  IF p_id_personal IS NOT NULL THEN
    UPDATE administracion.personal SET id_taller = p_id_taller WHERE id_personal = p_id_personal;
  END IF;
END $$;

        INSERT INTO viajes.ruta (id_servicio, nombre_ruta) VALUES (p_id_servicio, p_nombre_ruta) RETURNING id_ruta INTO p_id_ruta;

        DELETE FROM viajes.ruta_paradas WHERE id_ruta = p_id_ruta;

CREATE OR REPLACE PROCEDURE ventas.sp_registrar_venta(INOUT p_id_venta INT, INOUT p_folio INT,
    p_id_app INT, p_id_turno INT, p_id_viaje INT, p_id_origen INT, p_id_destino INT,
    p_nombre TEXT, p_ap_paterno TEXT, p_ap_materno TEXT, p_curp TEXT,
    p_asientos INT[], p_tarifas INT[], p_recibido NUMERIC, p_reemplaza_a INT DEFAULT NULL)
LANGUAGE plpgsql SECURITY DEFINER SET search_path = pg_catalog, pg_temp AS $$
DECLARE
  v_rol TEXT; v_taquilla INT; v_activo BOOLEAN;
  v_t administracion.taquilla%ROWTYPE;
  v_id_ruta INT; v_estado TEXT; v_oo INT; v_od INT; v_precio NUMERIC;
  v_total NUMERIC; v_pasajero INT; v_n INT;
BEGIN
  SELECT r.nombre_rol, p.id_taquilla, (pa.activo AND p.activo) INTO v_rol, v_taquilla, v_activo
  FROM administracion.personal_app pa JOIN administracion.personal p ON p.id_personal = pa.id_personal
  JOIN administracion.rol r ON r.id_rol = pa.id_rol WHERE pa.id_app = p_id_app;
  IF NOT FOUND OR NOT v_activo OR v_rol <> 'TAQUILLERO' THEN
    RAISE EXCEPTION 'Solo un taquillero activo puede vender.' USING ERRCODE = 'Z0040';
  END IF;
  IF v_taquilla IS NULL THEN RAISE EXCEPTION 'El taquillero no tiene taquilla asignada.' USING ERRCODE = 'Z0040'; END IF;
  v_n := COALESCE(array_length(p_asientos, 1), 0);
  IF v_n = 0 OR v_n <> COALESCE(array_length(p_tarifas, 1), 0) THEN
    RAISE EXCEPTION 'Indique un asiento y una tarifa por boleto.' USING ERRCODE = 'Z0044';
  END IF;
  IF (SELECT count(DISTINCT a) FROM unnest(p_asientos) a) <> v_n THEN
    RAISE EXCEPTION 'Hay asientos repetidos en la venta.' USING ERRCODE = 'Z0044';
  END IF;
  SELECT * INTO v_t FROM administracion.taquilla WHERE id_taquilla = v_taquilla FOR UPDATE;
  IF NOT v_t.activo THEN RAISE EXCEPTION 'La taquilla está inactiva.' USING ERRCODE = 'Z0040'; END IF;
  SELECT vp.id_ruta, ev.nombre_estado_viaje INTO v_id_ruta, v_estado
  FROM viajes.viaje_programado vp JOIN viajes.estado_viaje ev ON ev.id_estado_viaje = vp.id_estado_viaje WHERE vp.id_viaje = p_id_viaje;
  IF NOT FOUND THEN RAISE EXCEPTION 'El viaje no existe.' USING ERRCODE = 'Z0042'; END IF;
  IF v_estado = 'CERRADO' THEN RAISE EXCEPTION 'El viaje ya está cerrado.' USING ERRCODE = 'Z0042'; END IF;
  SELECT orden INTO v_oo FROM viajes.ruta_paradas WHERE id_ruta = v_id_ruta AND id_terminal = p_id_origen;
  SELECT orden INTO v_od FROM viajes.ruta_paradas WHERE id_ruta = v_id_ruta AND id_terminal = p_id_destino;
  SELECT precio INTO v_precio FROM viajes.precio_tramo WHERE id_ruta = v_id_ruta AND orden_origen = v_oo AND orden_destino = v_od;
  IF v_precio IS NULL THEN RAISE EXCEPTION 'No hay precio definido para ese tramo.' USING ERRCODE = 'Z0042'; END IF;
  IF p_reemplaza_a IS NOT NULL THEN
    CALL ventas.sp_cancelar_venta(p_reemplaza_a, p_id_app);
  END IF;
  SELECT COALESCE(sum(round(v_precio * (1 - t.descuento), 2)), 0), count(t.id_tarifa) INTO v_total, v_n
  FROM unnest(p_tarifas) x JOIN ventas.tarifa t ON t.id_tarifa = x;
  IF v_n <> array_length(p_tarifas, 1) THEN RAISE EXCEPTION 'Hay una tarifa que no existe.' USING ERRCODE = 'Z0044'; END IF;
  IF p_recibido < v_total THEN
    RAISE EXCEPTION 'El dinero recibido (%) no cubre el total (%).', p_recibido, v_total USING ERRCODE = 'Z0043';
  END IF;
  p_folio := v_t.rango_inicial + v_t.folios_usados;
  IF p_folio > v_t.rango_final THEN
    RAISE EXCEPTION 'La taquilla agotó su rango de folios.' USING ERRCODE = 'Z0041';
  END IF;
  IF p_curp IS NOT NULL AND btrim(p_curp) <> '' THEN
    INSERT INTO ventas.pasajero (nombre_pasajero, ap_paterno, ap_materno, curp)
    VALUES (btrim(p_nombre), btrim(p_ap_paterno), btrim(p_ap_materno), upper(btrim(p_curp)))
    ON CONFLICT (curp) DO UPDATE SET nombre_pasajero = EXCLUDED.nombre_pasajero,
      ap_paterno = EXCLUDED.ap_paterno, ap_materno = EXCLUDED.ap_materno
    RETURNING id_pasajero INTO v_pasajero;
  ELSE
    INSERT INTO ventas.pasajero (nombre_pasajero, ap_paterno, ap_materno)
    VALUES (btrim(p_nombre), btrim(p_ap_paterno), btrim(p_ap_materno)) RETURNING id_pasajero INTO v_pasajero;
  END IF;
  INSERT INTO ventas.venta (id_taquilla, folio, id_titular, id_vendedor, id_turno, total_pagado, recibido, reemplaza_a)
  VALUES (v_taquilla, p_folio, v_pasajero, p_id_app, p_id_turno, v_total, p_recibido, p_reemplaza_a)
  RETURNING id_venta INTO p_id_venta;
  INSERT INTO ventas.boleto (id_venta, id_viaje, id_terminal_origen, id_terminal_destino, num_asiento,
                             tramo_ocupacion, id_tarifa, id_estado_boleto, precio_unitario)
  SELECT p_id_venta, p_id_viaje, p_id_origen, p_id_destino, a.asiento, int4range(v_oo, v_od, '[)'), a.tarifa, 1,
         round(v_precio * (1 - t.descuento), 2)
  FROM unnest(p_asientos, p_tarifas) AS a(asiento, tarifa) JOIN ventas.tarifa t ON t.id_tarifa = a.tarifa;
  UPDATE administracion.taquilla SET folios_usados = folios_usados + 1 WHERE id_taquilla = v_taquilla;
END $$;


CREATE PROCEDURE ventas.sp_cancelar_venta(p_id_venta INT, p_id_app INT)
LANGUAGE plpgsql SECURITY DEFINER SET search_path = pg_catalog, pg_temp AS $$
DECLARE
  v_cancelada TIMESTAMPTZ;
  v_rol TEXT;
BEGIN
  SELECT r.nombre_rol INTO v_rol FROM administracion.personal_app pa JOIN administracion.rol r ON r.id_rol = pa.id_rol
  WHERE pa.id_app = p_id_app AND pa.activo;
  IF v_rol IS NULL OR v_rol NOT IN ('TAQUILLERO', 'ADMINISTRADOR') THEN
    RAISE EXCEPTION 'No tiene permiso para cancelar ventas.' USING ERRCODE = 'Z0040';
  END IF;
  SELECT cancelada_en INTO v_cancelada FROM ventas.venta WHERE id_venta = p_id_venta FOR UPDATE;
  IF NOT FOUND THEN RAISE EXCEPTION 'La venta no existe.' USING ERRCODE = 'Z0045'; END IF;
  IF v_cancelada IS NOT NULL THEN RAISE EXCEPTION 'La venta ya estaba cancelada.' USING ERRCODE = 'Z0045'; END IF;
  IF EXISTS (SELECT 1 FROM ventas.boleto WHERE id_venta = p_id_venta AND id_estado_boleto = 3) THEN
    RAISE EXCEPTION 'La venta tiene pasajeros que ya abordaron.' USING ERRCODE = 'Z0045';
  END IF;
  UPDATE ventas.boleto SET id_estado_boleto = 4 WHERE id_venta = p_id_venta;
  UPDATE ventas.venta SET cancelada_por = p_id_app, cancelada_en = now() WHERE id_venta = p_id_venta;
END $$;



CREATE OR REPLACE PROCEDURE ventas.sp_marcar_abordado(p_id_boleto INT)
LANGUAGE plpgsql
AS $$
BEGIN
UPDATE ventas.boleto 
SET id_estado_boleto = (SELECT id_estado_boleto FROM ventas.estado_boleto WHERE nombre_estado_boleto = 'ABORDADO')
WHERE id_boleto = p_id_boleto;
END;
$$;
