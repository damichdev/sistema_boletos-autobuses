CREATE OR REPLACE FUNCTION viajes.fn_obtener_escalas(p_id_ruta INT)
RETURNS TABLE (
id_terminal INT,

) 
LANGUAGE sql
AS $$
SELECT t.id_terminal, t.nombre_terminal, rp.min_desde_origen
FROM viajes.ruta_paradas rp

CREATE OR REPLACE VIEW ventas.v_manifiesto_detallado AS
SELECT 
b.id_viaje,
b.id_terminal_origen,
b.num_asiento, 
p.nombre_pasajero || ' ' || p.ap_paterno || COALESCE(' ' || p.ap_materno, '') AS nombre_completo, 
td.nombre_terminal AS destino,
tar.nombre_tarifa, 
upper(q.serie_folio || '-' || lpad(v.folio::text, 6, '0')) AS folio_impreso,
CASE 
WHEN eb.nombre_estado_boleto = 'ABORDADO' THEN 'Abordó'
WHEN ev.nombre_estado_viaje IN ('EN RUTA', 'CERRADO')
AND vp.fecha_salida + make_interval(mins => rp.min_desde_origen) <= now() THEN 'No se presentó'
ELSE 'Pendiente' 
END AS estatus_abordaje,
eb.nombre_estado_boleto
FROM ventas.boleto b
INNER JOIN ventas.venta v ON v.id_venta = b.id_venta
INNER JOIN ventas.pasajero p ON p.id_pasajero = v.id_titular
INNER JOIN administracion.taquilla q ON q.id_taquilla = v.id_taquilla
INNER JOIN administracion.terminal td ON td.id_terminal = b.id_terminal_destino
INNER JOIN ventas.tarifa tar ON tar.id_tarifa = b.id_tarifa
INNER JOIN ventas.estado_boleto eb ON eb.id_estado_boleto = b.id_estado_boleto
INNER JOIN viajes.viaje_programado vp ON vp.id_viaje = b.id_viaje
INNER JOIN viajes.estado_viaje ev ON ev.id_estado_viaje = vp.id_estado_viaje
INNER JOIN viajes.ruta_paradas rp ON rp.id_ruta = vp.id_ruta AND rp.id_terminal = b.id_terminal_origen;


CREATE OR REPLACE FUNCTION ventas.fn_buscar_pasajeros_manifiesto(p_id_viaje INT, p_id_terminal INT)
RETURNS TABLE (
num_asiento INT, 
nombre_completo TEXT, 
destino TEXT, 
nombre_tarifa TEXT, 
folio_impreso TEXT, 
estatus_abordaje TEXT
) 
LANGUAGE sql
AS $$
SELECT 
v.num_asiento, 
v.nombre_completo, 
v.destino, 
v.nombre_tarifa, 
v.folio_impreso, 
v.estatus_abordaje
FROM ventas.v_manifiesto_detallado v
WHERE v.id_viaje = p_id_viaje 
AND v.id_terminal_origen = p_id_terminal 
AND v.nombre_estado_boleto <> 'CANCELADO'
ORDER BY v.num_asiento;
$$;

CREATE OR REPLACE FUNCTION ventas.fn_obtener_asientos_ocupados(p_id_viaje INT, p_id_origen INT, p_id_destino INT)
RETURNS TABLE (
num_asiento INT
)
LANGUAGE sql
AS $$
SELECT b.num_asiento
FROM ventas.boleto b
WHERE b.id_viaje = p_id_viaje
AND b.id_estado_boleto <> 4
AND b.tramo_ocupacion && (
SELECT int4range(o.orden, d.orden, '[)')
FROM viajes.viaje_programado vp
INNER JOIN viajes.ruta_paradas o ON o.id_ruta = vp.id_ruta AND o.id_terminal = p_id_origen
INNER JOIN viajes.ruta_paradas d ON d.id_ruta = vp.id_ruta AND d.id_terminal = p_id_destino
WHERE vp.id_viaje = p_id_viaje
)
ORDER BY b.num_asiento;
$$;

-- HORARIOS DE RUTA (administrador, desde el catalogo de rutas)
-- Las horas de salida oficiales se vinculan a una ruta ya creada; la lista se lee con fn_obtener_horarios_ruta.

CREATE OR REPLACE PROCEDURE viajes.sp_agregar_horario_ruta(p_id_ruta INT, p_hora_salida TIME)
LANGUAGE plpgsql

CREATE OR REPLACE PROCEDURE viajes.sp_quitar_horario_ruta(p_id_ruta INT, p_hora_salida TIME)
LANGUAGE plpgsql
AS $$
BEGIN
DELETE FROM viajes.horario_ruta WHERE id_ruta = p_id_ruta AND hora_salida = p_hora_salida;
END;
$$;
