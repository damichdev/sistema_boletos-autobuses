-- EXTENSIONES Y CONFIGURACIONES

ALTER DATABASE taller SET timezone TO 'America/Mexico_City';

-- ////////////////////////////////////////////////////////////////////////////////////////////////////


-- TABLAS

CREATE TABLE estado_taller(
id_estado INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
nombre_estado TEXT NOT NULL UNIQUE
);


CREATE TABLE bus_taller(
placa TEXT primary key,
id_estado INT NOT NULL DEFAULT 1,
ultima_fecha_ingreso TIMESTAMPTZ,
observaciones TEXT,
CONSTRAINT fk_estado_bus FOREIGN KEY (id_estado) REFERENCES estado_taller(id_estado) ON DELETE RESTRICT ON UPDATE CASCADE
);

CREATE TABLE mecanico(
id_mecanico INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
nombre_mecanico TEXT NOT NULL,
ap_paterno TEXT NOT NULL,
ap_materno TEXT,
especialidad TEXT NOT NULL,
activo BOOLEAN DEFAULT TRUE
);

CREATE TABLE orden_completa(
id_orden INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
placa TEXT NOT NULL,
id_mecanico INT NOT NULL,
fecha_ingreso TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
fecha_salida TIMESTAMPTZ,
diagnostico TEXT,
costo DECIMAL(10,2) NOT NULL,
CONSTRAINT check_fecha_salida CHECK (fecha_salida IS NULL OR fecha_salida >= fecha_ingreso),
CONSTRAINT check_costo_positivo CHECK (costo >= 0),
CONSTRAINT fk_placa_orden FOREIGN KEY (placa) REFERENCES bus_taller(placa) ON DELETE RESTRICT ON UPDATE CASCADE,
CONSTRAINT fk_mecanico_orden FOREIGN KEY (id_mecanico) REFERENCES mecanico(id_mecanico) ON DELETE RESTRICT ON UPDATE CASCADE
);

-- VISTA

CREATE OR REPLACE VIEW v_semaforo_unidades AS
SELECT
b.placa,
e.nombre_estado AS estatus,
CASE
WHEN e.id_estado = 1 THEN TRUE
ELSE FALSE
END AS apto_para_viajar
FROM bus_taller b
INNER JOIN estado_taller e ON b.id_estado = e.id_estado;
