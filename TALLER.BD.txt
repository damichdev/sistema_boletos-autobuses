-- TALLER.BD

-- EXTENSIONES Y CONFIGURACIONES

ALTER DATABASE taller SET timezone TO 'America/Mexico_City';

CREATE SCHEMA taller;

ALTER DATABASE taller SET search_path = taller, public;

-- ////////////////////////////////////////////////////////////////////////////////////////////////////


-- TABLAS

CREATE TABLE taller.estado_taller(
id_estado INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
nombre_estado TEXT NOT NULL UNIQUE
);


CREATE TABLE taller.bus_taller(
placa TEXT primary key,
id_estado INT NOT NULL DEFAULT 1,
id_taller INT,
ultima_fecha_ingreso TIMESTAMPTZ,
observaciones TEXT,
CONSTRAINT fk_estado_bus FOREIGN KEY (id_estado) REFERENCES taller.estado_taller(id_estado) ON DELETE RESTRICT ON UPDATE CASCADE
);


CREATE TABLE taller.orden_completa(
id_orden INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
placa TEXT NOT NULL,
id_taller INT NOT NULL,
id_personal INT NOT NULL,
fecha_ingreso TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
fecha_salida TIMESTAMPTZ,
diagnostico TEXT,
costo DECIMAL(10,2) NOT NULL,
CONSTRAINT check_fecha_salida CHECK (fecha_salida IS NULL OR fecha_salida >= fecha_ingreso),
CONSTRAINT check_costo_positivo CHECK (costo >= 0),
CONSTRAINT fk_placa_orden FOREIGN KEY (placa) REFERENCES taller.bus_taller(placa) ON DELETE RESTRICT ON UPDATE CASCADE
);

CREATE UNIQUE INDEX unq_orden_abierta ON taller.orden_completa (placa) WHERE fecha_salida IS NULL;

-- VISTA

CREATE OR REPLACE VIEW taller.v_semaforo_unidades AS
SELECT
b.placa,
e.nombre_estado AS estatus,
CASE
WHEN e.id_estado = 1 THEN TRUE
ELSE FALSE
END AS apto_para_viajar
FROM taller.bus_taller b
INNER JOIN taller.estado_taller e ON b.id_estado = e.id_estado;
